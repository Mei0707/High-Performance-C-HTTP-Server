cmake_minimum_required(VERSION 3.20)
project(PulseHTTP VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "PulseHTTP targets Linux (epoll/eventfd). On macOS use Docker; see README.md.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PULSE_BUILD_TESTS "Build GoogleTest and integration tests" ON)
option(PULSE_BUILD_BENCHMARKS "Build Google Benchmark microbenchmarks" OFF)
option(PULSE_FUZZ "Build libFuzzer parser target (Clang)" OFF)
set(PULSE_SANITIZER "" CACHE STRING "Sanitizer: address,undefined or thread")
if(PULSE_SANITIZER)
  add_compile_options(-fsanitize=${PULSE_SANITIZER} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${PULSE_SANITIZER})
endif()
find_package(Threads REQUIRED)
add_library(pulsehttp
  src/config.cpp src/http_parser.cpp src/http_response.cpp src/router.cpp
  src/thread_pool.cpp src/metrics.cpp src/logger.cpp src/reactor.cpp src/server.cpp)
target_include_directories(pulsehttp PUBLIC include)
target_link_libraries(pulsehttp PUBLIC Threads::Threads)
target_compile_options(pulsehttp PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(pulsehttp_server src/main.cpp)
target_link_libraries(pulsehttp_server PRIVATE pulsehttp)
add_executable(custom_routes examples/custom_routes.cpp)
target_link_libraries(custom_routes PRIVATE pulsehttp)
add_executable(thread_per_connection benchmarks/baselines/thread_per_connection.cpp)
target_link_libraries(thread_per_connection PRIVATE pulsehttp)
if(PULSE_BUILD_TESTS)
  enable_testing()
  find_package(GTest REQUIRED)
  add_executable(unit_tests tests/test_http.cpp tests/test_components.cpp)
  target_link_libraries(unit_tests PRIVATE pulsehttp GTest::gtest_main)
  add_test(NAME unit COMMAND unit_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/test_server.py $<TARGET_FILE:pulsehttp_server>)
  add_test(NAME baseline_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/test_server.py $<TARGET_FILE:thread_per_connection> --baseline)
  add_test(NAME experiment_tools COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_experiment_tools.py)
  set_tests_properties(integration baseline_integration PROPERTIES TIMEOUT 100)
endif()
if(PULSE_BUILD_BENCHMARKS)
  find_package(benchmark REQUIRED)
  add_executable(parser_benchmark benchmarks/parser_benchmark.cpp)
  target_link_libraries(parser_benchmark PRIVATE pulsehttp benchmark::benchmark_main)
endif()
if(PULSE_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "PULSE_FUZZ requires Clang")
  endif()
  target_compile_options(pulsehttp PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(http_parser_fuzz fuzz/http_parser_fuzz.cpp)
  target_compile_options(http_parser_fuzz PRIVATE -fsanitize=fuzzer)
  target_link_options(http_parser_fuzz PRIVATE -fsanitize=fuzzer)
  target_link_libraries(http_parser_fuzz PRIVATE pulsehttp)
endif()
